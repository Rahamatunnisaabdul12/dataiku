def gethmac_request)signature(self):
"""
to create the signature
1. cocncate the values of fields listed bellow into a single string(no space/characters in between  any of value
fileds:(user ID/request/Uri/Times stamp/nonce/request body)
2. convet the string generated in steo 1 to UTF8
3. use the HMACSHA256 hashing algorithm to hash the string, passing the secrete key
4.convert the hash to base 64 encoding.
"""
#uri is first encode in UTF-8 format and then with percentage encoding
#quote module of urllib.parse does bith the encoding
encoded_uri=quote(self.uri, safe='').lower()

#generate an MD5 of the request body & encode it as a base 64 string
if self.request_params:
md5_hash= hashlib.md5(self.request_params).digest()
encode_request_params = b64encode (md5_hash).decode()

#Signature is created following the 4 steps mentioned in the function doc string
signature = f"(self.userid){self.method}{encoded_uri}{self.timestamp]{self.nonce} {encode_request_params}"
signature_utf8= signature.encode()
hashed_signature= hmac.new(self.secret_key, signature_utf8, hashlib.sha256).digest()
hashed_base64 b64encode (hashed_signature).decode()
return hashed_base64
